```cpp
  const double speed_gap = std::fabs(inp.own_ship.u_mps - inp.planned_speed_mps);
  if (speed_gap > max_speed_delta) {
    spdlog::warn("[M5][MidMPC] speed gap {:.1f} m/s exceeds N·decel_max·dt {:.1f}; flagging",
                 speed_gap, max_speed_delta);
```

```cpp
// types.hpp（或新头文件 mid_mpc/speed_contract.hpp）
inline bool compute_speed_gap_infeasible(
    double own_u_mps, double planned_u_mps,
    double decel_max_mps2, int n_horizon, double dt_s) {
  const double max_delta = decel_max_mps2 * dt_s * static_cast<double>(n_horizon);
  return std::fabs(own_u_mps - planned_u_mps) > max_delta;
}
```

```cpp
// colregs_directive.cpp 实现：
HeadingBoxReachability compute_heading_box_reachability(
    double h_min_deg, double h_max_deg,
    double own_hdg_deg, double rot_step_deg,
    double min_alt_rad) {
  HeadingBoxReachability r;
  // box 起点距 own_psi（own-relative 坐标，取靠 own 的边）
  const double d_min = signed_delta_deg(own_hdg_deg, h_min_deg);
  const double d_max = signed_delta_deg(own_hdg_deg, h_max_deg);
  // 哪个边靠 own_psi 近且在 directive 方向上（sign 与 directive 同）
  // 简化：取 |d_min|, |d_max| 较小者作为 reachable_from_psi0
  const double reach_deg = std::min(std::fabs(d_min), std::fabs(d_max));
  r.heading_box_reachable_from_psi0_deg = reach_deg;

  const double min_alt_deg = min_alt_rad * 180.0 / M_PI;
  // box 是否允许 min_alt 量级偏转：box 宽度 ≥ min_alt_deg + rot_step_deg
  const double box_width = std::fabs(h_max_deg - h_min_deg);
  r.box_allows_min_alt = (box_width >= min_alt_deg + rot_step_deg);
  if (!r.box_allows_min_alt) {
    r.reachability_rationale = "box_width < min_alt + rot_step";
  }
  return r;
}
```

```cpp
namespace mass_l3::m4 {

double WeightedSumCombination::combine(
    double psi_deg, double u_kn,
    const std::vector<WeightedFunction>& fns) const {
  double result = 0.0;
  for (const auto& wf : fns) {
    result += wf.weight * wf.function.evaluate(psi_deg, u_kn);
  }
  return result;
}
```

```cpp
      {0.3, fn2}

  EXPECT_NEAR(result, 0.7 * 0.6 + 0.3 * 0.4, 1e-6);  // 0.42 + 0.12 = 0.54
```

```cpp
IvPHardConstraints IvPHardConstraints::relax(int level) const {
  IvPHardConstraints out = *this;
  switch (level) {
    case 0:  // Original — no relaxation
      break;
    case 1:  // Remove CPA constraints (targets + cpa_safe)
      out.targets.clear();
      out.cpa_safe_m = 0.0;
      break;
    case 2:  // Remove heading constraints
      out.heading_allowed_ranges_deg.clear();
      break;
    case 3:  // Speed-only bounds (heading also cleared here if called)
      out.heading_allowed_ranges_deg.clear();
      break;
    default:
      break;
  }
  return out;
}
```

```cpp
std::optional<IvPSolution> IvPSolver::solve_with_fallback(
    const std::vector<IvPCombinationStrategy::WeightedFunction>& weighted_fns,
    const IvPHardConstraints& constraints) const {
  // Try original constraints
  auto sol = solve(weighted_fns, constraints);
  if (sol.has_value()) {
    sol->relax_level = 0;
    return sol;
  }

  // Cascading relaxation
  for (int level = 1; level <= 3; ++level) {
    IvPHardConstraints relaxed = constraints.relax(level);
    sol = solve(weighted_fns, relaxed);
    if (sol.has_value()) {
      sol->relax_level = level;
      return sol;
    }
  }

  return std::nullopt;
}
```

```cpp
#include "m4_behavior_arbiter/behavior_activation.hpp"

#include <algorithm>
#include <spdlog/spdlog.h>

namespace mass_l3::m4 {

HealthState BehaviorActivationCondition::compute_health_state(
    const ArbitrationInputs& inputs) {
  int stale_count = 0;
  if (inputs.age_odd_ms     > inputs.timeout_critical_ms) ++stale_count;
  if (inputs.age_world_ms   > inputs.timeout_critical_ms) ++stale_count;
  if (inputs.age_mission_ms > inputs.timeout_critical_ms) ++stale_count;
  if (inputs.age_colregs_ms > inputs.timeout_critical_ms) ++stale_count;

  if (stale_count >= 3) return HealthState::Critical;

  bool any_degraded =
      (inputs.age_odd_ms     > inputs.timeout_degraded_ms) ||
      (inputs.age_world_ms   > inputs.timeout_degraded_ms) ||
      (inputs.age_mission_ms > inputs.timeout_degraded_ms) ||
      (inputs.age_colregs_ms > inputs.timeout_degraded_ms);

  return any_degraded ? HealthState::Degraded : HealthState::Normal;
}

bool BehaviorActivationCondition::is_transit_applicable(
    const ArbitrationInputs& inputs) {
  // Active in ODD-A, B, C zones
  if (inputs.odd_zone > 2) return false;  // not ODD-D (3) or out-of-range
  // Must have mission goal received
  if (!inputs.mission_received) return false;
  return true;
}

bool BehaviorActivationCondition::is_colreg_avoid_applicable(
    const ArbitrationInputs& inputs) {
  // Active in ODD-A, B, D (not C — harbor)
  if (inputs.odd_zone == 2) return false;  // ODD-C
  // COLREGs constraint must be fresh
  if (!inputs.colregs_received) return false;
  if (inputs.age_colregs_ms > inputs.timeout_degraded_ms) return false;
  // Conflict must be detected
  if (!inputs.colregs_conflict_detected) return false;
  return true;
}

bool BehaviorActivationCondition::is_restricted_vis_applicable(
    const ArbitrationInputs& inputs) {
  // Only in ODD-D (restricted visibility)
  if (inputs.odd_zone != 3) return false;
  // Visibility must be below 2 nm
  if (!inputs.world_received) return false;
  if (inputs.world_visibility_nm >= 2.0) return false;
  // Ship must be moving
  if (inputs.own_speed_kn <= 0.0) return false;
  return true;
}

bool BehaviorActivationCondition::is_channel_follow_applicable(
    const ArbitrationInputs& inputs) {
  // Only in ODD-B (narrow channel)
  if (inputs.odd_zone != 1) return false;
  // Must be in a VTS zone
  if (!inputs.world_received) return false;
  if (!inputs.world_in_vts_zone) return false;
  return true;
}

bool BehaviorActivationCondition::is_mrc_drift_applicable(
    const ArbitrationInputs& inputs) {
  // MRC triggered by M1 mode command
  if (inputs.mode_mrc_triggered) return true;
  // All inputs timed out (critical health)
  HealthState hs = compute_health_state(inputs);
  if (hs == HealthState::Critical) return true;
  return false;
}

std::vector<BehaviorType> BehaviorActivationCondition::compute_active_set(
    const ArbitrationInputs& inputs,
    const BehaviorDictionary& dictionary) {
  std::vector<BehaviorType> active;

  // MRC_Drift is checked first — it overrides everything else
  if (is_mrc_drift_applicable(inputs)) {
    active.push_back(BehaviorType::MrcDrift);
    return active;  // MRC is exclusive
  }

  if (is_transit_applicable(inputs)) {
    active.push_back(BehaviorType::Transit);
  }

  if (is_colreg_avoid_applicable(inputs)) {
    active.push_back(BehaviorType::ColregAvoid);
  }

  if (is_restricted_vis_applicable(inputs)) {
    active.push_back(BehaviorType::DpHold);  // enum slot 2 = RESTRICTED_VIS in 5-behavior model
  }

  if (is_channel_follow_applicable(inputs)) {
    active.push_back(BehaviorType::Berth);   // enum slot 3 = CHANNEL_FOLLOW in 5-behavior model
  }

  spdlog::debug("[M4] compute_active_set: {} behaviors active", active.size());
  return active;
}

}  // namespace mass_l3::m4
```

> **NOTE:** The 5-behavior model uses `BehaviorType` enum slots as: Transit(0), ColregAvoid(1), RestrictedVis=DpHold(2), ChannelFollow=Berth(3), MrcDrift(4). This is intentional — the enum names are 7-behavior legacy from v1.1.2; D3.1 activates only 0-4 with 5-behavior semantics.

```cpp
// --- Main arbitration callback ---
void BehaviorArbiterNode::arbitration_timer_callback() {
  // Step 1: Input snapshot
  ArbitrationInputs inputs = build_inputs();

  // Step 2: Health state
  HealthState health = BehaviorActivationCondition::compute_health_state(inputs);

  // Standby: no critical inputs
  if (!odd_received_ || !world_received_) {
    BehaviorPlanMsg plan;
    plan.schema_version = 113;
    plan.stamp = now();
    plan.behavior = BehaviorPlanMsg::BEHAVIOR_TRANSIT;
    plan.heading_min_deg = 0.0f;
    plan.heading_max_deg = 360.0f;
    plan.speed_min_kn = 0.0f;
    plan.speed_max_kn = 22.0f;
    plan.confidence = 0.0f;
    plan.rationale = "Standby: waiting for inputs";
    pub_plan_->publish(plan);
    return;
  }

  // Step 3: Behavior activation
  auto active_set = BehaviorActivationCondition::compute_active_set(inputs, dictionary_);

  // Step 4: Check for MRC override (after activation)
  bool has_mrc = false;
  for (auto b : active_set) {
    if (b == BehaviorType::MrcDrift) { has_mrc = true; break; }
  }

  // --- Build result ---
  BehaviorType primary = BehaviorType::MrcDrift;
  double h_min = 0.0, h_max = 359.0, s_min = 0.0, s_max = 22.0;
  double confidence = 0.95;
  std::string rationale;

  if (has_mrc) {
    // MRC override — skip IvP, output emergency drift
    primary = BehaviorType::MrcDrift;
    h_min = 0.0; h_max = 360.0;
    s_min = 0.0; s_max = 0.0;  // stop propulsion
    confidence = 1.0;
    rationale = "MRC_DRIFT override: emergency drift active";
  } else if (!active_set.empty()) {
    // Normal IvP arbitration path
    // Build weighted functions (stub: empty for now, utility functions are E2+)
    std::vector<IvPCombinationStrategy::WeightedFunction> weighted_fns;
    // TODO(D3.1 E2): Build actual IvP functions per behavior

    IvPHardConstraints constraints;
    constraints.speed_min_kn = 0.0;
    constraints.speed_max_kn = 22.0;

    // Inject M6 COLREGs heading constraints
    if (colregs_received_ && latest_colregs_ && latest_colregs_->conflict_detected) {
      for (const auto& c : latest_colregs_->constraints) {
        if (c.direction == COLREGsConstraintMsg::STARBOARD) {
          constraints.heading_allowed_ranges_deg.push_back(
              {latest_world_->own_ship.heading_deg + c.min_action_deg,
               latest_world_->own_ship.heading_deg + 180.0});
        }
      }
    }

    auto sol = solver_->solve_with_fallback(weighted_fns, constraints);

    // Select primary behavior
    if (!active_set.empty()) {
      // Priority: first non-MRC behavior by default, COLREGs wins if active
      primary = active_set[0];
      for (auto b : active_set) {
        if (b == BehaviorType::ColregAvoid) { primary = b; break; }
      }
    }

    if (sol.has_value()) {
      h_min = sol->heading_min_deg;
      h_max = sol->heading_max_deg;
      s_min = sol->speed_min_kn;
      s_max = sol->speed_max_kn;
      confidence = sol->relax_level > 0 ? 0.75 : 0.95;
      rationale = sol->rationale;
    } else {
      // Conservative fallback
      h_min = latest_world_ ? latest_world_->own_ship.heading_deg - 90.0 : 0.0;
      h_max = latest_world_ ? latest_world_->own_ship.heading_deg + 90.0 : 360.0;
      s_max = latest_world_ ? 0.5 * latest_world_->own_ship.sog_kn : 10.0;
      confidence = 0.30;
      rationale = "IvP infeasible fallback";
    }
  } else {
    // No active behaviors — standby transit
    primary = BehaviorType::Transit;
    confidence = 0.60;
    rationale = "No active behaviors; default Transit";
  }

  // Step 8: Publish Behavior_PlanMsg
  BehaviorPlanMsg plan;
  plan.schema_version = 113;
  plan.stamp = now();
  plan.behavior = static_cast<uint8_t>(primary);
  plan.heading_min_deg = static_cast<float>(h_min);
  plan.heading_max_deg = static_cast<float>(h_max);
  plan.speed_min_kn = static_cast<float>(s_min);
  plan.speed_max_kn = static_cast<float>(s_max);
  plan.confidence = static_cast<float>(confidence);
  plan.rationale = rationale;
  pub_plan_->publish(plan);

  // --- Publish ivp_contributions (SAT-2) ---
  // 8 compass directions with utility values
  static const double kDirections[8] = {0.0, 45.0, 90.0, 135.0, 180.0, 225.0, 270.0, 315.0};
  if (!has_mrc) {
    // Build a simple SAT-2 contribution message using BehaviorPlanMsg as interim carrier
    // D3.1 stub: use the plan fields to encode contributions
    BehaviorPlanMsg sat2;
    sat2.schema_version = 113;
    sat2.stamp = now();
    sat2.behavior = static_cast<uint8_t>(primary);
    // Encode 8-direction utility as plan fields (hack until SAT2Data.msg gets ivp_contributions)
    sat2.heading_min_deg = 8.0f;  // indicate "contribution mode" with count=8
    sat2.heading_max_deg = static_cast<float>(confidence);
    sat2.speed_min_kn = static_cast<float>(h_min);
    sat2.speed_max_kn = static_cast<float>(h_max);
    sat2.confidence = static_cast<float>(confidence);
    sat2.rationale = rationale;
    pub_sat2_->publish(sat2);
  }

  // Step 9: ASDR
  // Behavior change event
  if (primary != prev_primary_) {
    std::ostringstream json;
    json << "{\"from\":" << static_cast<int>(prev_primary_)
         << ",\"to\":" << static_cast<int>(primary)
         << ",\"active_count\":" << active_set.size()
         << ",\"odd_zone\":" << static_cast<int>(inputs.odd_zone) << "}";
    publish_asdr_event("behavior_switch", json.str());
  }

  // ODD transition event
  if (prev_odd_zone_ != 99 && inputs.odd_zone != prev_odd_zone_) {
    std::ostringstream json;
    json << "{\"from_zone\":" << static_cast<int>(prev_odd_zone_)
         << ",\"to_zone\":" << static_cast<int>(inputs.odd_zone) << "}";
    publish_asdr_event("odd_transition", json.str());
  }

  prev_primary_ = primary;
  prev_odd_zone_ = inputs.odd_zone;
  prev_health_ = health;
}

double BehaviorArbiterNode::compute_confidence(
    const ArbitrationResult& result) const {
```
```cpp
  double base = result.confidence;  // set by arbitration logic
  // freshness_factor

  auto now = this->now();
  if (odd_received_ && latest_odd_) {

    if (age > 2000) stale_count++;

  if (world_received_ && latest_world_) {

    if (age > 2000) stale_count++;
  }
  if (stale_count == 1) freshness = 0.8;

  return std::clamp(base * freshness, 0.0, 1.0);
```

#ifndef MASS_L3_M5_MID_MPC_NOMOTO_FALLBACK_HPP_
#define MASS_L3_M5_MID_MPC_NOMOTO_FALLBACK_HPP_

// M5 Tactical Planner — Nomoto 1st-order fallback solver (DEMO-2 P0).
// Used when M5_USE_CASADI=OFF or IPOPT repeatedly times out.
//
// Nomoto model:  T·ṙ + r = K·δ
// Discrete (Forward Euler): r_{k+1} = r_k + (dt/T)*(K·δ - r_k)
//   ψ_{k+1} = ψ_k + r_k·dt
//   x_{k+1} = x_k + u·cos(ψ_k)·dt
//   y_{k+1} = y_k + u·sin(ψ_k)·dt

#include <cstdint>
#include <vector>

#include <Eigen/Dense>

#include "m5_tactical_planner/common/types.hpp"
#include "m5_tactical_planner/shared/capability_manifest.hpp"

namespace mass_l3::m5::mid_mpc {

struct NomotoFallbackConfig {
  int32_t n_steps{12};      // N=12 for DEMO-2 (short horizon)
  double dt_s{5.0};         // step duration [s]
  int32_t n_branches{13};   // candidate heading branches (±60°)
  double delta_psi_rad{10.0 * 0.0174533};  // 10° step
};

struct NomotoFallbackSolution {
  /// For each branch i (0..n_branches-1): trajectory[i] has N+1 points.
  std::vector<std::vector<Eigen::Vector2d>> trajectories;
  std::vector<double> cpa_vals;   // linear CPA per branch [m]
  std::vector<double> headings_rad;  // branch heading [rad]
  int32_t primary_branch_idx{6};   // index of center (current heading) branch
};

class NomotoFallback {
 public:
  explicit NomotoFallback(const NomotoFallbackConfig& cfg,
                          const CapabilityManifest& manifest);

  /// Generate n_branches trajectories using Nomoto model at constant heading.
  /// Each branch holds heading constant (δ=0 after initial step), CPA computed
  /// with CpaCalculator::compute_linear.
  [[nodiscard]] NomotoFallbackSolution solve(const MidMpcInput& input) const;

 private:
  NomotoFallbackConfig cfg_;
  double nomoto_T_s_;
  double nomoto_K_inv_s_;

  /// Single-branch forward integration: return N+1 NED positions.
  [[nodiscard]] std::vector<Eigen::Vector2d> integrate_branch(
      double x0_m, double y0_m, double psi_rad, double u_mps) const;
};

}  // namespace mass_l3::m5::mid_mpc

#endif  // MASS_L3_M5_MID_MPC_NOMOTO_FALLBACK_HPP_
```

#include "m5_tactical_planner/mid_mpc/nomoto_fallback.hpp"

#include <cmath>
#include <cstdint>
#include <vector>

#include <Eigen/Dense>

#include "m5_tactical_planner/common/types.hpp"
#include "m5_tactical_planner/shared/capability_manifest.hpp"
#include "m5_tactical_planner/shared/cpa_calculator.hpp"

namespace mass_l3::m5::mid_mpc {

using mass_l3::m5::shared::CpaCalculator;

NomotoFallback::NomotoFallback(const NomotoFallbackConfig& cfg,
                               const CapabilityManifest& manifest)
    : cfg_(cfg),
      nomoto_T_s_(manifest.config().nomoto_T_s),
      nomoto_K_inv_s_(manifest.config().nomoto_K_inv_s) {}

std::vector<Eigen::Vector2d> NomotoFallback::integrate_branch(
    double x0_m, double y0_m, double psi_rad, double u_mps) const {
  const int32_t N = cfg_.n_steps;
  std::vector<Eigen::Vector2d> positions;
  positions.reserve(static_cast<std::size_t>(N + 1));
  positions.emplace_back(x0_m, y0_m);

  // Nomoto: constant heading → δ=0, r decays to 0
  double r_rad_s = 0.0;  // start from zero turning rate
  double x = x0_m, y = y0_m;
  double psi = psi_rad;

  for (int32_t k = 0; k < N; ++k) {
    const double dt = cfg_.dt_s;
    // δ = 0 (hold heading); r decays: r_{k+1} = r_k - (dt/T)*r_k
    const double r_next = r_rad_s - (dt / nomoto_T_s_) * r_rad_s;
    r_rad_s = r_next;

    // Forward Euler integration
    psi += r_rad_s * dt;
    x   += u_mps * std::cos(psi) * dt;
    y   += u_mps * std::sin(psi) * dt;

    positions.emplace_back(x, y);
  }
  return positions;
}

NomotoFallbackSolution NomotoFallback::solve(const MidMpcInput& input) const {
  NomotoFallbackSolution sol;
  const int32_t nb = cfg_.n_branches;
  const double psi0 = input.own_ship.psi_rad;
  const double u0   = input.own_ship.u_mps;
  const double x0   = input.own_ship.x_m;
  const double y0   = input.own_ship.y_m;

  sol.trajectories.reserve(static_cast<std::size_t>(nb));
  sol.cpa_vals.reserve(static_cast<std::size_t>(nb));
  sol.headings_rad.reserve(static_cast<std::size_t>(nb));

  const int32_t half = nb / 2;  // 6 for nb=13

  for (int32_t i = 0; i < nb; ++i) {
    const double psi_branch = psi0 + cfg_.delta_psi_rad * static_cast<double>(i - half);
    sol.headings_rad.push_back(psi_branch);

    auto traj = integrate_branch(x0, y0, psi_branch, u0);
    sol.trajectories.push_back(std::move(traj));
  }

  // Linear CPA for each branch (DEMO-2: simple CPA, no intent-min)
  // Create a synthetic own-ship TrajectoryPoint at the branch heading
  TrajectoryPoint own_pt = input.own_ship;
  for (int32_t i = 0; i < nb; ++i) {
    own_pt.psi_rad = sol.headings_rad[static_cast<std::size_t>(i)];
    double min_cpa = 1.0e9;
    for (const auto& tgt : input.targets) {
      const auto result = CpaCalculator::compute_linear(own_pt, tgt);
      if (result.cpa_m < min_cpa) { min_cpa = result.cpa_m; }
    }
    if (input.targets.empty()) { min_cpa = 1.0e9; }
    sol.cpa_vals.push_back(min_cpa);
  }

  // Primary branch = center (i = half); current heading
  sol.primary_branch_idx = half;
  return sol;
}

}  // namespace mass_l3::m5::mid_mpc

TEST(NomotoFallbackTest, IntegrateStraightLine) {
  mass_l3::m5::mid_mpc::NomotoFallbackConfig cfg;
  cfg.n_steps = 12;
  cfg.dt_s = 5.0;
  cfg.n_branches = 13;

  mass_l3::m5::shared::CapabilityManifest::Config mc;
  mc.nomoto_T_s = 15.0;
  mc.nomoto_K_inv_s = 0.08;
  mass_l3::m5::shared::CapabilityManifest manifest(

          M5_TEST_FIXTURE_DIR "/fcb_capability_fixture.yaml"));

  mass_l3::m5::mid_mpc::NomotoFallback fallback(cfg, manifest);

  mass_l3::m5::MidMpcInput input;
  input.own_ship.x_m = 0.0;
  input.own_ship.y_m = 0.0;
  input.own_ship.psi_rad = 0.0;  // north
  input.own_ship.u_mps = 5.0;

  auto sol = fallback.solve(input);

  // 13 branches expected
  EXPECT_EQ(sol.trajectories.size(), 13u);
  EXPECT_EQ(sol.cpa_vals.size(), 13u);
  EXPECT_EQ(sol.headings_rad.size(), 13u);

  // Primary branch is center (idx 6); heading ≈ 0°
  EXPECT_EQ(sol.primary_branch_idx, 6);
  EXPECT_NEAR(sol.headings_rad[6], 0.0, 1e-6);

  // Center branch: N=12 steps × 5s = 60s @ 5 m/s → ~300m north
  const auto& primary_traj = sol.trajectories[6];
  EXPECT_EQ(primary_traj.size(), 13u);  // N+1 points
  EXPECT_NEAR(primary_traj.back().x(), 5.0 * 12 * 5.0, 50.0);  // ~300m north
  EXPECT_NEAR(primary_traj.back().y(), 0.0, 10.0);  // ~0 east
}

std::vector<double> BcMpcBranchFormulation::candidate_headings(
    double current_heading_rad, double urgency_level) const {
  std::int32_t k;
  if (urgency_level > 0.95) {
    k = cfg_.k_critical;   // ±60° for imminent collision (P2-B-01)
  } else if (urgency_level > cfg_.urgency_threshold) {
    k = cfg_.k_high;       // ±30° (default)
  } else {
    k = cfg_.k_low;        // ±20° (low urgency)
  }

  std::vector<double> headings;
  headings.reserve(static_cast<std::size_t>(k));
  const int32_t half = k / 2;
  for (int32_t i = 0; i < k; ++i) {
    const double offset = cfg_.delta_psi_rad * static_cast<double>(i - half);
    double psi = current_heading_rad + offset;
    // Normalize to [-π, π)
    while (psi >= M_PI)  { psi -= 2.0 * M_PI; }
    while (psi < -M_PI)  { psi += 2.0 * M_PI; }
    headings.push_back(psi);
  }
  return headings;
}

std::vector<Eigen::Vector2d> BcMpcBranchFormulation::predict_target_with_intent(
    const TargetState& target, int32_t intent_scenario) const {
  const int32_t N = num_steps_();
  std::vector<Eigen::Vector2d> positions;
  positions.reserve(static_cast<std::size_t>(N + 1));
  positions.emplace_back(target.x_m, target.y_m);

  double x = target.x_m, y = target.y_m;
  double cog = target.cog_rad;
  double sog = target.sog_mps;
  const double dt = cfg_.dt_s;

  for (int32_t k = 0; k < N; ++k) {
    switch (intent_scenario) {
      case 1:  // TurnPort: -3°/s
        cog -= 3.0 * M_PI / 180.0 * dt;
        break;
      case 2:  // TurnStarboard: +3°/s
        cog += 3.0 * M_PI / 180.0 * dt;
        break;
      case 3:  // Decelerate: -0.1 m/s²
        sog = std::max(0.0, sog - 0.1 * dt);
        break;
      default:  // Maintain: no change
        break;
    }
    x += sog * std::cos(cog) * dt;
    y += sog * std::sin(cog) * dt;
    positions.emplace_back(x, y);
  }
  return positions;
}

    const double prev_u = prev.trajectory[0].u_mps;
    const double curr_u = current.own_ship.u_mps;
    if (std::abs(curr_u - prev_u) > 3.0) {
      return false;
    }

    // Condition 3: heading change ≤ 15° (= 0.2618 rad)
    double delta_psi = std::abs(current.own_ship.psi_rad - prev.trajectory[0].psi_rad);
    if (delta_psi > M_PI) { delta_psi = 2.0 * M_PI - delta_psi; }
    if (delta_psi > 0.2618) {
      return false;
    }
  }

  // Condition 4: target IDs unchanged
  if (prev.status == MidMpcSolution::Status::Converged) {
    // If current input has different target count, invalidate
    // (exact ID comparison deferred to Phase E2 when target IDs tracked)
    if (current.targets.empty() != (prev.targets.empty())) {
      return false;
    }
  } else {
    // Solution didn't converge → invalid anyway
    return false;
  }

  // Condition 5: solution status is Converged (already checked above)
  return true;
}
```

void MidMpcNode::on_solve_cycle_()
{
  if (!has_required_inputs_()) {
    spdlog::warn("[M5][MidMPC] Skipping cycle: missing required inputs");
    return;
  }

  const MidMpcInput input = assemble_input_();

  // Always publish trajectory_candidates (Nomoto fallback for DEMO-2)
  publish_trajectory_candidates_(input);

  if (fallback_state_ == FallbackState::FALLBACK) {
    // In fallback: run Nomoto only, track recovery
    ++rrstar_good_cycles_;
    if (rrstar_good_cycles_ >= kRrtStarRecoveryCycles) {
      spdlog::info("[M5][MidMPC] RRT* recovery threshold reached, restoring Mid-MPC");
      fallback_state_ = FallbackState::NORMAL;
      consecutive_timeouts_ = 0;
      rrstar_good_cycles_ = 0;
    }
    // Still publish AvoidancePlan with status = DEGRADED (no Mid-MPC solve)
    l3_msgs::msg::AvoidancePlan degraded_plan;
    degraded_plan.stamp = this->get_clock()->now();
    degraded_plan.status = "DEGRADED";
    pub_avoidance_plan_->publish(degraded_plan);
    return;
  }

  // NORMAL state: run CasADi/IPOPT
  const MidMpcSolution* warm = last_solution_.has_value() ? &last_solution_.value() : nullptr;
  const MidMpcSolution sol = solver_.solve(input, warm);
  last_solution_ = sol;

  // Timeout check
  if (sol.solve_duration_ms > kTimeoutThreshold_ms) {
    ++consecutive_timeouts_;
    spdlog::warn("[M5][MidMPC] IPOPT timeout: {}ms (consecutive: {}/3)",
                 sol.solve_duration_ms, consecutive_timeouts_);

    if (consecutive_timeouts_ >= kConsecutiveTimeoutTrigger) {
      spdlog::error("[M5][MidMPC] 3 consecutive timeouts → triggering Nomoto/RRT* fallback");
      fallback_state_ = FallbackState::FALLBACK;
      rrstar_good_cycles_ = 0;
      // MRM-01 escalation: if fallback persists > 1.0s, M7 triggers deceleration
      // (M7 reads /m5/avoidance_plan status field)
    }
  } else {
    consecutive_timeouts_ = 0;  // success → reset counter
  }

  const double lat = world_state_->own_ship.position.latitude;
  const double lon = world_state_->own_ship.position.longitude;
  const auto plan = wp_gen_.generate(sol, lat, lon);

  // Override: mark DEGRADED if in fallback
  if (fallback_state_ == FallbackState::FALLBACK) {
    plan.status = "DEGRADED";
  }

  publish_outputs_(sol, plan);
}

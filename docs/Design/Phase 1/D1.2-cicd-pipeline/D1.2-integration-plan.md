```bash
# ============================================================
# 5. LOC Ratio Check (ARCH §11.x: M7:Doer ≥ 100:1)
# ============================================================
readonly DOER_DIRS=(
    "src/l3_tdl_kernel/m1_odd_envelope_manager"
    "src/l3_tdl_kernel/m2_world_model"
    "src/l3_tdl_kernel/m3_mission_manager"
    "src/l3_tdl_kernel/m4_behavior_arbiter"
    "src/l3_tdl_kernel/m5_tactical_planner"
    "src/l3_tdl_kernel/m6_colregs_reasoner"
)

if command -v cloc &>/dev/null; then
    M7_LOC=$(cloc --csv --quiet "${M7_SRC}" 2>/dev/null | tail -1 | cut -d, -f5)
    DOER_LOC=0
    for d in "${DOER_DIRS[@]}"; do
        [[ -d "${d}" ]] || continue
        pkg_loc=$(cloc --csv --quiet "${d}" 2>/dev/null | tail -1 | cut -d, -f5)
        DOER_LOC=$((DOER_LOC + pkg_loc))
    done
    M7_LOC=${M7_LOC:-0}
    DOER_LOC=${DOER_LOC:-1}  # avoid div-by-zero
    LOC_RATIO=$((DOER_LOC / (M7_LOC > 0 ? M7_LOC : 1)))
    echo "PATH-S LOC: M7=${M7_LOC} lines, Doers(M1-M6)=${DOER_LOC} lines, Ratio=${LOC_RATIO}:1 (target ≥100:1)"
    if [[ ${LOC_RATIO} -lt 100 ]]; then
        echo "VIOLATION: Doer/Checker LOC ratio ${LOC_RATIO}:1 < 100:1 threshold"
        errors=$((errors + 1))
    fi
else
    echo "WARNING: cloc not installed — skipping LOC ratio check"
fi
```

- [ ] **Step 2.3: Add cyclomatic complexity ratio check**

Append after the LOC section:

```bash
# ============================================================
# 6. Cyclomatic Complexity Ratio (ARCH §11.x: M7:Doer ≥ 30:1)
#    Uses average CC per function across codebase.
# ============================================================
if command -v lizard &>/dev/null; then
    # lizard outputs: NLOC  CCN  token  param  length  location
    # We extract column 2 (CCN) and compute averages.
    M7_CCN_AVG=$(lizard "${M7_SRC}" 2>/dev/null | awk '/^Total nloc/ {split($0,a," "); for(i=1;i<=length(a);i++) if(a[i]=="CCN") {print a[i+1]; exit}}')
    DOER_CCN_SUM=0; DOER_FUNC_COUNT=0
    for d in "${DOER_DIRS[@]}"; do
        [[ -d "${d}" ]] || continue
        ccn_val=$(lizard "${d}" 2>/dev/null | awk '/^Total nloc/ {split($0,a," "); for(i=1;i<=length(a);i++) if(a[i]=="CCN") {print a[i+1]; exit}}')
        funcs=$(lizard "${d}" 2>/dev/null | awk '/^Total nloc/ {split($0,a," "); for(i=1;i<=length(a);i++) if(a[i]=="avg.NLOC") {print a[i+1]; exit}}')
        [[ -n "${ccn_val}" ]] && DOER_CCN_SUM=$(echo "${DOER_CCN_SUM} + ${ccn_val}" | bc)
        DOER_FUNC_COUNT=$((DOER_FUNC_COUNT + 1))
    done
    M7_CCN_AVG=${M7_CCN_AVG:-0}
    DOER_CCN_AVG=$(echo "scale=1; ${DOER_CCN_SUM} / (${DOER_FUNC_COUNT} > 0 ? ${DOER_FUNC_COUNT} : 1)" | bc 2>/dev/null || echo 0)
    CC_RATIO=$(echo "scale=1; ${DOER_CCN_AVG} / (${M7_CCN_AVG} > 0 ? ${M7_CCN_AVG} : 1)" | bc 2>/dev/null || echo 0)
    echo "PATH-S CCN: M7_avg=${M7_CCN_AVG}, Doer_avg=${DOER_CCN_AVG}, Ratio=${CC_RATIO}:1 (target ≥30:1)"
    if (( $(echo "${CC_RATIO} < 30" | bc -l) )); then
        echo "VIOLATION: Doer/Checker CC ratio ${CC_RATIO}:1 < 30:1 threshold"
        errors=$((errors + 1))
    fi
else
    echo "WARNING: lizard not installed — skipping cyclomatic complexity ratio check"
fi
```

```bash
# ============================================================
# 7. SBOM Intersection Check (ARCH §11.x: M7 ∩ Doer deps = ∅)
# ============================================================
if command -v syft &>/dev/null; then
    TMPDIR=$(mktemp -d)
    trap "rm -rf ${TMPDIR}" EXIT

    # Generate SBOM for M7
    syft dir:"${M7_SRC}" -o cyclonedx-json --quiet > "${TMPDIR}/m7_sbom.json" 2>/dev/null || true

    # Generate merged SBOM for Doers
    DOER_PATHS=""
    for d in "${DOER_DIRS[@]}"; do
        [[ -d "${d}" ]] && DOER_PATHS="${DOER_PATHS} ${d}"
    done
    syft dir:${DOER_PATHS} -o cyclonedx-json --quiet > "${TMPDIR}/doer_sbom.json" 2>/dev/null || true

    # Extract library names from both SBOMs and check intersection
    if [[ -s "${TMPDIR}/m7_sbom.json" && -s "${TMPDIR}/doer_sbom.json" ]]; then
        M7_LIBS=$(jq -r '.components[]?.name // empty' "${TMPDIR}/m7_sbom.json" 2>/dev/null | sort -u)
        DOER_LIBS=$(jq -r '.components[]?.name // empty' "${TMPDIR}/doer_sbom.json" 2>/dev/null | sort -u)
        INTERSECTION=$(comm -12 <(echo "${M7_LIBS}") <(echo "${DOER_LIBS}") 2>/dev/null)
        if [[ -n "${INTERSECTION}" ]]; then
            echo "VIOLATION: Shared SBOM components between M7 and Doers:"
            echo "${INTERSECTION}" | while read -r lib; do echo "  - ${lib}"; done
            errors=$((errors + 1))
        else
            echo "SBOM intersection: ∅ (M7 and Doers share no dependencies)"
        fi
    else
        echo "WARNING: SBOM generation incomplete — skipping intersection check"
    fi
else
    echo "INFO: syft not installed — skipping SBOM intersection check (non-blocking, DEMO-1 contingency)"
fi
```

# src/sim_workbench/sil_nodes/ship_dynamics/ship_dynamics/ship_motion_simulator.py
"""ShipMotionSimulator 抽象基类 — pluginlib 风格的 Python plugin 契约。

对应 C++ ship_sim_interfaces::ShipMotionSimulator，Python 实现使用 abc.ABC。
D1.3.3 FMI bridge 消费 export_fmu_interface() 返回值生成 modelDescription.xml。
"""

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from .mmg_model import ShipState  # 复用既有 ShipState


@dataclass
class FmuVariableSpec:
    """单个 FMI 2.0 变量描述 — 对应 FMI modelDescription.xml <ScalarVariable>。"""
    name: str
    causality: str        # "input" | "output" | "parameter" | "local"
    variability: str      # "continuous" | "discrete" | "fixed" | "tunable"
    type: str             # "Real" | "Integer" | "Boolean" | "String"
    unit: str             # "m/s", "rad", "rad/s", "1" (dimensionless)
    start: float = 0.0    # 初始值 (Real 类型)
    description: str = "" # ≤120 chars


@dataclass
class FmuDescriptor:
    """FMI 2.0 模型描述容器 — export_fmu_interface() 返回值。

    D1.3.3 消费路径:
        1. 遍历 variables[] → 生成 modelDescription.xml
        2. causality=="input" 变量 ↔ ROS2 subscriber topic
        3. causality=="output" 变量 ↔ ROS2 publisher topic
    """
    fmi_version: str = "2.0"
    model_name: str = ""
    model_identifier: str = ""
    default_step_size: float = 0.02
    variables: list[FmuVariableSpec] = field(default_factory=list)


class ShipMotionSimulator(ABC):
    """船舶运动仿真器抽象基类。

    所有 vessel plugin 必须实现此接口。ShipDynamicsNode 通过
    YAML vessel_class 字段动态加载具体 plugin，不硬编码船型。
    """

    @abstractmethod
    def load_params(self, yaml_path: str) -> None:
        """从 YAML 文件加载船型特定参数 (Capability Manifest)。"""

    @abstractmethod
    def step(
        self,
        state: ShipState,
        delta_rad: float,
        n_rps: float,
        dt_s: float,
        wind_speed: float = 0.0,
        wind_dir_rad: float = 0.0,
        current_speed: float = 0.0,
        current_dir_rad: float = 0.0,
    ) -> ShipState:
        """单步积分: 输入当前状态 + 控制/环境量，输出 dt_s 后的新状态。"""

    @abstractmethod
    def vessel_class(self) -> str:
        """返回 vessel class 标识符 (如 "FCB")。"""

    @abstractmethod
    def hull_class(self) -> str:
        """返回 hull class 标识符 (如 "SEMI_PLANING")。"""

    @abstractmethod
    def export_fmu_interface(self) -> FmuDescriptor:
        """导出 FMI 2.0 模型描述 (D1.3.3 消费)。

        Phase 1 仅签名 + FCBPlugin stub 实装。D1.3.3 消费
        FmuDescriptor 生成 modelDescription.xml + dds-fmu mapping。
        新 vessel plugin 只需填变量清单，无业务逻辑。
        """
```

# src/sim_workbench/sil_nodes/ship_dynamics/ship_dynamics/fcb_plugin.py
"""FCBPlugin — FCB 45m semi-planing patrol vessel 动力学 plugin。

包装现有 MMGModel (Yasukawa 2015 4-DOF)，实现 ShipMotionSimulator 契约。
零逻辑 fork: 所有动力学计算由 MMGModel 完成，本类仅做参数加载 + 接口适配。
"""

from __future__ import annotations
import math
from .mmg_model import MMGModel, ShipState
from .mmg_coefficients import MMGCoefficients
from .ship_motion_simulator import (
    ShipMotionSimulator,
    FmuDescriptor,
    FmuVariableSpec,
)


class FCBPlugin(ShipMotionSimulator):
    """FCB 45m 半规划巡逻艇动力学 plugin (Yasukawa 2015 [R7])。

    hull_class = SEMI_PLANING, Fn = 0.45–0.63。
    """

    def __init__(self):
        self._model: MMGModel | None = None
        self._coeffs: MMGCoefficients | None = None

    # ── ShipMotionSimulator 契约 ────────────────────────────

    def load_params(self, yaml_path: str) -> None:
        import yaml
        with open(yaml_path) as f:
            cfg = yaml.safe_load(f)
        params = cfg.get("ship_dynamics", {}).get("ros__parameters", {})

        coeffs = MMGCoefficients()
        # 从 YAML 逐字段赋值（保留 dataclass 默认值作 fallback）
        for field_name in coeffs.__dataclass_fields__:
            if field_name in params:
                setattr(coeffs, field_name, params[field_name])

        self._coeffs = coeffs
        self._model = MMGModel(coeffs)

    def step(
        self,
        state: ShipState,
        delta_rad: float,
        n_rps: float,
        dt_s: float,
        wind_speed: float = 0.0,
        wind_dir_rad: float = 0.0,
        current_speed: float = 0.0,
        current_dir_rad: float = 0.0,
    ) -> ShipState:
        if self._model is None:
            raise RuntimeError("FCBPlugin: load_params() must be called before step()")
        return self._model.rk4_step(
            state, delta_rad, n_rps,
            wind_speed, wind_dir_rad, current_speed, current_dir_rad,
        )

    def vessel_class(self) -> str:
        return "FCB"

    def hull_class(self) -> str:
        return "SEMI_PLANING"

    def export_fmu_interface(self) -> FmuDescriptor:
        return FmuDescriptor(
            fmi_version="2.0",
            model_name="FCBShipDynamics",
            model_identifier="FCBShipDynamics",
            default_step_size=0.02,
            variables=[
                # ── outputs (own-ship state, 11) ──────────
                FmuVariableSpec("u",     "output", "continuous", "Real", "m/s",   0.0, "surge velocity (body frame)"),
                FmuVariableSpec("v",     "output", "continuous", "Real", "m/s",   0.0, "sway velocity (body frame)"),
                FmuVariableSpec("r",     "output", "continuous", "Real", "rad/s", 0.0, "yaw rate"),
                FmuVariableSpec("x",     "output", "continuous", "Real", "m",     0.0, "position x (NED)"),
                FmuVariableSpec("y",     "output", "continuous", "Real", "m",     0.0, "position y (NED)"),
                FmuVariableSpec("psi",   "output", "continuous", "Real", "rad",   0.0, "heading"),
                FmuVariableSpec("phi",   "output", "continuous", "Real", "rad",   0.0, "roll angle"),
                FmuVariableSpec("p",     "output", "continuous", "Real", "rad/s", 0.0, "roll rate"),
                FmuVariableSpec("delta", "output", "continuous", "Real", "rad",   0.0, "actual rudder angle"),
                FmuVariableSpec("n",     "output", "continuous", "Real", "rev/s", 0.0, "actual propeller rps"),
                FmuVariableSpec("sog",   "output", "continuous", "Real", "m/s",   0.0, "speed over ground"),
                # ── control inputs (commands, 2) ──────────
                FmuVariableSpec("delta_cmd", "input", "continuous", "Real", "rad",   0.0, "commanded rudder angle"),
                FmuVariableSpec("n_rps_cmd", "input", "continuous", "Real", "rev/s", 0.0, "commanded propeller rps"),
                # ── disturbance inputs, 4 ──────────────────
                FmuVariableSpec("wind_dir_deg",      "input", "continuous", "Real", "deg",  0.0, "true wind direction (from)"),
                FmuVariableSpec("wind_speed_mps",    "input", "continuous", "Real", "m/s",  0.0, "true wind speed"),
                FmuVariableSpec("current_dir_deg",   "input", "continuous", "Real", "deg",  0.0, "current set direction (towards)"),
                FmuVariableSpec("current_speed_mps", "input", "continuous", "Real", "m/s",  0.0, "current speed"),
                # ── parameters (FCB hydro static, 5) ──────
                FmuVariableSpec("L",   "parameter", "fixed", "Real", "m",      45.0,  "ship length"),
                FmuVariableSpec("B",   "parameter", "fixed", "Real", "m",       8.5,  "beam"),
                FmuVariableSpec("d",   "parameter", "fixed", "Real", "m",       2.5,  "draft"),
                FmuVariableSpec("m",   "parameter", "fixed", "Real", "kg",  3.5e5,   "mass"),
                FmuVariableSpec("Izz", "parameter", "fixed", "Real", "kg.m2",5.0e7,  "yaw inertia"),
            ],
        )
```

def _load_plugin(self) -> ShipMotionSimulator:
    """根据 YAML vessel_class 字段动态加载 ShipMotionSimulator plugin。

    plugin 注册表:
        "FCB" → FCBPlugin
    新增 vessel 时只需添加注册项 + 对应的 plugin 模块。
    严禁在 Python 代码中硬编码 "FCB" 等船型字面量
    （multi_vessel_lint 合规——vessel_class 值从 YAML 读取）。
    """
    # 从参数服务器读取 vessel_class
    try:
        vessel_class = self.get_parameter("vessel_class").get_parameter_value().string_value
    except Exception:
        vessel_class = "FCB"  # fallback

    # Plugin 注册表 (单一映射点，multi_vessel_lint 白名单豁免)
    _PLUGIN_REGISTRY: dict[str, type[ShipMotionSimulator]] = {
        "FCB": FCBPlugin,
        # "TUG": TugPlugin,      # RUN-002 拖船 (Phase 4)
        # "FERRY": FerryPlugin,  # RUN-003 渡船 (Phase 4)
    }

    plugin_cls = _PLUGIN_REGISTRY.get(vessel_class)
    if plugin_cls is None:
        raise ValueError(
            f"Unknown vessel_class '{vessel_class}'. "
            f"Available: {list(_PLUGIN_REGISTRY.keys())}"
        )

    plugin = plugin_cls()

    # 定位 YAML 配置文件路径
    import os
    config_dir = os.path.join(
        os.path.dirname(os.path.dirname(__file__)), "config"
    )
    yaml_path = os.path.join(config_dir, "fcb_dynamics.yaml")

    plugin.load_params(yaml_path)
    return plugin
```

#!/usr/bin/env python3

"""DEMO-1 D1.3.1 集成 launch: ShipDynamicsNode + AISReplayNode。

启动后 30s 内两个 topic 均有数据:
  /sil/own_ship_state @ 50Hz
  /sil/target_vessel_state @ 2Hz (AIS 回放)

from launch import LaunchDescription
from launch_ros.actions import Node
from launch.actions import DeclareLaunchArgument
from launch.substitutions import LaunchConfiguration


def generate_launch_description():
    replay_rate = LaunchConfiguration('replay_rate_x', default='5')

    return LaunchDescription([
        DeclareLaunchArgument(
            'replay_rate_x', default_value='5',
            description='AIS replay speed multiplier (1=real-time, 5=5x, 10=10x)'
        ),

        # ShipDynamicsNode — FCB 4-DOF MMG @ 50Hz
        Node(
            package='ship_dynamics',
            executable='ship_dynamics_node',
            name='ship_dynamics_node',
            parameters=['src/sim_workbench/sil_nodes/ship_dynamics/config/fcb_dynamics.yaml'],
            output='screen',
        ),

        # AISReplayNode — NOAA/DMA AIS 历史回放
        Node(
            package='ais_bridge',
            executable='ais_replay_node',
            name='ais_replay_node',
            parameters=[{
                'dataset_path': 'data/ais_datasets/AIS_synthetic_1h.csv',
                'dataset_format': 'noaa_csv',
                'replay_rate_x': replay_rate,
                'publish_rate_hz': 2.0,
                'max_targets': 50,
            }],
            output='screen',
        ),
    ])
```

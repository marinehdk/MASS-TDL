```python
"""Per-rule sub-criteria evaluator for COLREGs rules 13/14/15/16/17.

D1.7 §5.3 rubric. Returns "full" | "partial" | "violated" given encounter_state dict.

encounter_state keys (all optional with defaults):
  heading_change_deg  float  absolute heading change in degrees
  cpa_nm              float  achieved CPA in nautical miles
  cpa_target_nm       float  minimum required CPA (default 0.27 NM)
  rudder_side         str    "starboard" | "port"
  role                str    "give_way" | "stand_on"
  acted_early         bool   True if action taken at STAGE_1 or STAGE_2
  timing_stage        str    "STAGE_1" | "STAGE_2" | "STAGE_3" | "EMERGENCY"
"""
from __future__ import annotations


def evaluate_rule_compliance(rule: str, encounter_state: dict) -> str:
    """Return "full" | "partial" | "violated" for *rule* given *encounter_state*."""
    handlers = {
        "Rule13": _eval_rule13,
        "Rule14": _eval_rule14,
        "Rule15": _eval_rule15,
        "Rule16": _eval_rule16,
        "Rule17": _eval_rule17,
    }
    handler = handlers.get(rule)
    if handler is None:
        return "full"  # rules not in scope (R5/6/7/8/9/19) pass through
    return handler(encounter_state)


def _eval_rule13(s: dict) -> str:
    """Rule 13 Overtaking: give-way vessel must keep clear."""
    hc = abs(s.get("heading_change_deg", 0.0))
    cpa = s.get("cpa_nm", 0.0)
    cpa_target = s.get("cpa_target_nm", 0.27)
    if hc >= 30.0 and cpa >= cpa_target:
        return "full"
    if hc >= 15.0:
        return "partial"
    return "violated"


def _eval_rule14(s: dict) -> str:
    """Rule 14 Head-on: both vessels must alter to starboard."""
    side = s.get("rudder_side", "starboard")
    if side == "port":
        return "violated"
    hc = abs(s.get("heading_change_deg", 0.0))
    cpa = s.get("cpa_nm", 0.0)
    cpa_target = s.get("cpa_target_nm", 0.27)
    if hc >= 30.0 and cpa >= cpa_target:
        return "full"
    if hc >= 15.0:
        return "partial"
    return "violated"


def _eval_rule15(s: dict) -> str:
    """Rule 15 Crossing: give-way passes astern; stand-on maintains course."""
    role = s.get("role", "give_way")
    hc = abs(s.get("heading_change_deg", 0.0))
    cpa = s.get("cpa_nm", 0.0)
    cpa_target = s.get("cpa_target_nm", 0.27)
    if role == "give_way":
        if hc >= 30.0 and cpa >= cpa_target:
            return "full"
        if hc >= 15.0:
            return "partial"
        return "violated"
    # stand_on: must maintain course and speed
    if hc < 5.0:
        return "full"
    if hc < 15.0:
        return "partial"
    return "violated"


def _eval_rule16(s: dict) -> str:
    """Rule 16 Give-way: action shall be early, large, and effective."""
    hc = abs(s.get("heading_change_deg", 0.0))
    cpa = s.get("cpa_nm", 0.0)
    cpa_target = s.get("cpa_target_nm", 0.27)
    early = s.get("acted_early", True)
    if hc >= 30.0 and cpa >= cpa_target and early:
        return "full"
    if hc >= 15.0:
        return "partial"
    return "violated"

    """Rule 17 Stand-on: maintain course/speed; independent action only at STAGE_3."""
    hc = abs(s.get("heading_change_deg", 0.0))
    stage = s.get("timing_stage", "STAGE_1")

            return "full"
        if hc < 15.0:
            return "partial"
        return "violated"  # turned when should hold
    # STAGE_3 or EMERGENCY: independent action permitted
    if hc < 30.0:
        return "partial"  # small action, still compliant
    return "partial"      # large independent action also partial (Rule 17b)
```

```python
class NcdmVessel(TargetShipReplayer):
    """NCDM Ornstein-Uhlenbeck stochastic vessel trajectory (D2.4).

    Generates a full OU-perturbed trajectory at init time for deterministic replay.
    """

    def __init__(
        self,
        lat0: float,
        lon0: float,
        heading0_deg: float,
        sog_kn: float,
        duration_s: float = 600.0,
        dt: float = 0.1,
        ou_theta: float = 0.05,
        ou_sigma: float = 0.5,
        seed: int | None = None,
    ) -> None:
        import math
        rng = np.random.default_rng(seed)
        n = int(duration_s / dt)
        headings = np.zeros(n)
        headings[0] = math.radians(heading0_deg)
        h_ref = headings[0]
        for i in range(1, n):
            dH = (
                -ou_theta * (headings[i - 1] - h_ref) * dt
                + ou_sigma * math.sqrt(dt) * rng.standard_normal()
            )
            headings[i] = headings[i - 1] + dH

        sog_ms = sog_kn * 0.514444
        lats = np.zeros(n)
        lons = np.zeros(n)
        lats[0] = lat0
        lons[0] = lon0
        for i in range(1, n):
            lat_r = math.radians(lats[i - 1])
            lats[i] = lats[i - 1] + sog_ms * math.cos(headings[i - 1]) * dt / 111120.0
            lons[i] = lons[i - 1] + sog_ms * math.sin(headings[i - 1]) * dt / (
                111120.0 * math.cos(lat_r)
            )

        t = np.arange(n) * dt
        cogs_deg = np.degrees(headings) % 360.0
        self._traj = np.column_stack([t, lats, lons, np.full(n, sog_kn), cogs_deg])
        self._t_end = float(t[-1])
        self._dt = dt

    def get_targets_at(self, t_s: float) -> dict | None:
        if t_s < 0.0 or t_s > self._t_end:
            return None
        idx = min(int(t_s / self._dt), len(self._traj) - 1)
        row = self._traj[idx]
        return {"lat": float(row[1]), "lon": float(row[2]),
                "sog_kn": float(row[3]), "cog_deg": float(row[4])}
```

```python
Reads AIS CSV (mmsi, timestamp_s, lat, lon, cog_deg, sog_kn) and finds
encounters satisfying DCPA ≤ 2.0 NM AND TCPA ∈ [0, 30 min].
Outputs maritime-schema YAML for each extracted encounter.

Usage:
    python tools/sil/extract_ais_encounters.py \
        --input data/ais_gunnerus.csv \
        --output scenarios/ais_derived/ \
        --max 5

If --input file does not exist, runs in synthetic-demo mode and writes
5 geometry-equivalent scenarios tagged source: synthetic_ais_equiv.

from __future__ import annotations

import argparse
import csv
import math
from datetime import datetime, timezone
from pathlib import Path

import yaml


R_NM = 3440.065  # Earth radius in nautical miles


def _haversine_nm(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    dlat = math.radians(lat2 - lat1)
    dlon = math.radians(lon2 - lon1)
    a = (math.sin(dlat / 2) ** 2
         + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2))
         * math.sin(dlon / 2) ** 2)
    return 2 * R_NM * math.atan2(math.sqrt(a), math.sqrt(1 - a))


def _relative_velocity(lat1: float, lon1: float, cog1: float, sog1: float,
                        lat2: float, lon2: float, cog2: float, sog2: float,
                        ) -> tuple[float, float]:
    """Return (dcpa_nm, tcpa_s) using CPA geometry in flat-Earth approximation."""
    dx_nm = (lon2 - lon1) * math.cos(math.radians((lat1 + lat2) / 2)) * 60.0
    dy_nm = (lat2 - lat1) * 60.0
    vx1 = sog1 * math.sin(math.radians(cog1))
    vy1 = sog1 * math.cos(math.radians(cog1))
    vx2 = sog2 * math.sin(math.radians(cog2))
    vy2 = sog2 * math.cos(math.radians(cog2))
    dvx = vx2 - vx1
    dvy = vy2 - vy1
    dv2 = dvx ** 2 + dvy ** 2
    if dv2 < 1e-9:
        return _haversine_nm(lat1, lon1, lat2, lon2), 0.0
    tcpa_h = -(dx_nm * dvx + dy_nm * dvy) / dv2
    tcpa_s = tcpa_h * 3600.0
    cx = dx_nm + dvx * tcpa_h
    cy = dy_nm + dvy * tcpa_h
    dcpa = math.sqrt(cx ** 2 + cy ** 2)
    return dcpa, tcpa_s


def _classify_encounter(bearing_deg: float) -> str:
    """Classify COLREGs encounter type from relative bearing."""
    b = bearing_deg % 360.0
    if b <= 15.0 or b >= 345.0:
        return "Rule14"
    if 15.0 < b <= 112.5:
        return "Rule15_starboard"
    if 112.5 < b <= 247.5:
        return "Rule13"
    return "Rule15_port"


def _bearing_deg(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    dlon = math.radians(lon2 - lon1)
    lat1r = math.radians(lat1)
    lat2r = math.radians(lat2)
    y = math.sin(dlon) * math.cos(lat2r)
    x = math.cos(lat1r) * math.sin(lat2r) - math.sin(lat1r) * math.cos(lat2r) * math.cos(dlon)
    return math.degrees(math.atan2(y, x)) % 360.0

def _run_synthetic(output_dir: Path, n: int) -> None:
    output_dir.mkdir(parents=True, exist_ok=True)
    counters: dict[str, int] = {}
    for i, (rule, ol, oo, oc, os_, tl, to, tc, ts) in enumerate(_SYNTHETIC_ENCOUNTERS[:n]):
        counters[rule] = counters.get(rule, 0) + 1
        bearing = _bearing_deg(ol, oo, tl, to)
        dcpa, _ = _relative_velocity(ol, oo, oc, os_, tl, to, tc, ts)
        os_dict = {"mmsi": 123456789, "lat": ol, "lon": oo, "cog": oc, "sog": os_}
        ts_dict = {"mmsi": 100000020 + i, "lat": tl, "lon": to, "cog": tc, "sog": ts}
        data = _make_scenario_yaml(counters[rule], rule, os_dict, ts_dict, dcpa,
                                    "synthetic_ais_equiv")
        stem = f"ais_{rule.lower().replace('_', '-')}_{counters[rule]:02d}"
        out = output_dir / f"{stem}.yaml"
        out.write_text(yaml.dump(data, allow_unicode=True, default_flow_style=False))
        print(f"  wrote {out}")


def run(input_csv: Path | None, output_dir: Path, max_scenarios: int) -> None:
    if input_csv is None or not input_csv.exists():
        print(f"[INFO] AIS CSV not found — using synthetic_ais_equiv mode")
        _run_synthetic(output_dir, max_scenarios)
        return

    rows: dict[int, list[dict]] = {}
    with open(input_csv) as f:
        for row in csv.DictReader(f):
            mmsi = int(row["mmsi"])
            rows.setdefault(mmsi, []).append({
                "t": float(row["timestamp_s"]),
                "lat": float(row["lat"]), "lon": float(row["lon"]),
                "cog": float(row["cog_deg"]), "sog": float(row["sog_kn"]),
            })

    mmsis = list(rows.keys())
    encounters = []
    for i, m1 in enumerate(mmsis):
        for m2 in mmsis[i + 1:]:
            r1 = rows[m1]
            r2 = rows[m2]
            for p1 in r1[::10]:
                for p2 in r2[::10]:
                    if abs(p1["t"] - p2["t"]) > 60:
                        continue
                    dcpa, tcpa = _relative_velocity(
                        p1["lat"], p1["lon"], p1["cog"], p1["sog"],
                        p2["lat"], p2["lon"], p2["cog"], p2["sog"])
                    if dcpa <= 2.0 and 0 <= tcpa <= 1800:
                        bearing = _bearing_deg(p1["lat"], p1["lon"], p2["lat"], p2["lon"])
                        rule = _classify_encounter(bearing)
                        encounters.append((rule, dcpa, p1, p2, m1, m2))
                        break
            if len(encounters) >= max_scenarios:
                break
        if len(encounters) >= max_scenarios:
            break

    output_dir.mkdir(parents=True, exist_ok=True)
    counters: dict[str, int] = {}
    for rule, dcpa, p1, p2, m1, m2 in encounters[:max_scenarios]:
        counters[rule] = counters.get(rule, 0) + 1
        os_d = {"mmsi": m1, "lat": p1["lat"], "lon": p1["lon"],
                "cog": p1["cog"], "sog": p1["sog"]}
        ts_d = {"mmsi": m2, "lat": p2["lat"], "lon": p2["lon"],
                "cog": p2["cog"], "sog": p2["sog"]}
        data = _make_scenario_yaml(counters[rule], rule, os_d, ts_d, dcpa,
                                    input_csv.name)
        stem = f"ais_{rule.lower().replace('_', '-')}_{counters[rule]:02d}"
        out = output_dir / f"{stem}.yaml"
        out.write_text(yaml.dump(data, allow_unicode=True, default_flow_style=False))
        print(f"  wrote {out}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Extract AIS encounter scenarios")
    parser.add_argument("--input", type=Path, default=None)
    parser.add_argument("--output", type=Path, default=Path("scenarios/ais_derived"))
    parser.add_argument("--max", type=int, default=5)
    args = parser.parse_args()
    run(args.input, args.output, args.max)
```

```cpp
ColregsReasonerNode::ColregsChainResult ColregsReasonerNode::build_colregs_chain(
    const std::vector<RuleEvaluation>& evals,
    OddDomain domain,
    const RuleParameters& params,
    const std::vector<TargetGeometricState>& targets) const {
  ColregsChainResult result;
  if (evals.empty() || targets.empty()) {
    return result;
  }

  // Primary target: minimum CPA among all targets
  const TargetGeometricState* primary_geo = nullptr;
  double min_cpa = std::numeric_limits<double>::max();
  for (const auto& t : targets) {
    if (t.cpa_m < min_cpa) {
      min_cpa = t.cpa_m;
      primary_geo = &t;
    }
  }
  if (!primary_geo) return result;

  // Primary evaluation: first active eval for primary target, else any active
  const RuleEvaluation* primary_eval = nullptr;
  for (const auto& ev : evals) {
    if (ev.is_active && ev.target_id == primary_geo->target_id) {
      primary_eval = &ev;
      break;
    }
  }
  if (!primary_eval) {
    for (const auto& ev : evals) {
      if (ev.is_active) { primary_eval = &ev; break; }
    }
  }
  if (!primary_eval) return result;

  result.target_id = std::to_string(primary_geo->target_id);
  result.layers.resize(5);

  // Layer 1: ODD domain
  result.layers[0] = make_chain_layer(
      1, "ODD", odd_domain_str(domain), {}, {}, 1.0f, 0, false,
      "Current ODD domain from M1");

  // Layer 2: Active rule set
  std::string active_rules;
  for (const auto& ev : evals) {
    if (ev.is_active && ev.target_id == primary_geo->target_id) {
      if (!active_rules.empty()) active_rules += " ";
      active_rules += "R" + std::to_string(ev.rule_id);
    }
  }
  result.layers[1] = make_chain_layer(
      2, "\xe8\xa7\x84\xe5\x88\x99\xe9\x9b\x86",
      active_rules.empty() ? "NONE" : active_rules,
      {}, {}, 0.9f, 0, false,
      "Active COLREGS rules for primary target");

  // Layer 3: Encounter classification
  const double kCpa_nm = primary_geo->cpa_m / 1852.0;
  result.layers[2] = make_chain_layer(
      3, "\xe4\xbc\x9a\xe9\x81\x87\xe5\x88\x86\xe7\xb1\xbb",
      encounter_type_str(primary_eval->encounter_type),
      {"bearing_deg", "cpa_nm"},
      {std::to_string(static_cast<int>(primary_geo->bearing_deg)),
       std::to_string(kCpa_nm).substr(0, 5)},
      primary_eval->confidence, 0, false,
      primary_eval->rationale);

  // Layer 4: Role assignment
  result.layers[3] = make_chain_layer(
      4, "\xe8\xb4\xa3\xe4\xbb\xbb\xe5\x88\x86\xe9\x85\x8d",
      role_str(primary_eval->role),
      {"aspect_deg"},
      {std::to_string(static_cast<int>(primary_geo->aspect_deg))},
      primary_eval->confidence, 0, false,
      "Role from M6 rule evaluation");

  // Layer 5: Timing phase
  const bool kEscalation =
      (primary_eval->phase == TimingPhase::INDEPENDENT_ACTION ||
       primary_eval->phase == TimingPhase::CRITICAL_ACTION);
  result.layers[4] = make_chain_layer(
      5, "\xe6\x97\xb6\xe6\x9c\xba",
      timing_phase_str(primary_eval->phase),
      {"tcpa_s", "t_act_s"},
      {std::to_string(static_cast<int>(primary_geo->tcpa_s)),
       std::to_string(static_cast<int>(params.t_act_s))},
      primary_eval->confidence,
      static_cast<uint8_t>(primary_eval->phase) + 1,
      kEscalation,
      "Timing phase based on TCPA vs ODD thresholds");

  return result;
}
```

```python
def _haversine_nm(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    R = 3440.065
    dlat = math.radians(lat2 - lat1)
    dlon = math.radians(lon2 - lon1)
    a = (math.sin(dlat / 2) ** 2
         + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2))
         * math.sin(dlon / 2) ** 2)
    return 2 * R * math.atan2(math.sqrt(a), math.sqrt(1 - a))


def _simulate_scenario_geometric(
    os_lat: float, os_lon: float, os_cog: float, os_sog: float,
    ts_lat: float, ts_lon: float, ts_cog: float, ts_sog: float,
    total_time_s: float = 600.0, dt: float = 1.0,
    rudder_deg: float = 45.0,
) -> tuple[float, list[dict]]:
    """Simplified geometric simulation — propagate positions, record CPA per timestep."""
    # Convert to flat-Earth (m) coordinates relative to origin
    def _to_xy(lat: float, lon: float, lat0: float, lon0: float) -> tuple[float, float]:
        x = (lon - lon0) * math.cos(math.radians(lat0)) * 111120.0
        y = (lat - lat0) * 111120.0
        return x, y

    lat0, lon0 = os_lat, os_lon
    ox, oy = 0.0, 0.0
    tx, ty = _to_xy(ts_lat, ts_lon, lat0, lon0)
    osog = os_sog * 0.514444
    tsog = ts_sog * 0.514444
    ocog = math.radians(os_cog)
    tcog = math.radians(ts_cog)

    min_dist_m = float("inf")
    frames = []
    action_taken = False

    for step in range(int(total_time_s / dt)):
        t = step * dt
        dist = math.sqrt((tx - ox) ** 2 + (ty - oy) ** 2)
        if dist < min_dist_m:
            min_dist_m = dist

        cpa_nm = dist / 1852.0
        rule_state = "full" if dist > 500.0 else "partial" if dist > 200.0 else "violated"
        frames.append({
            "stamp": t, "cpa_nm": cpa_nm, "rule14": rule_state,
            "rudder_deg": rudder_deg if action_taken else 0.0,
            "behavior_phase": "give_way" if action_taken else "transit",
        })
        # Trigger avoidance at 300s
        if t >= 300.0 and not action_taken:
            ocog = math.radians(os_cog + rudder_deg)
            action_taken = True

        ox += osog * math.sin(ocog) * dt
        oy += osog * math.cos(ocog) * dt
        tx += tsog * math.sin(tcog) * dt
        ty += tsog * math.cos(tcog) * dt

    return min_dist_m / 1852.0, frames


def run_scenario_yaml(yaml_path: Path, rule: str, source: str) -> ScenarioBatchResult:
    data = yaml.safe_load(yaml_path.read_text())
    meta = data.get("metadata", {})
    sid = meta.get("scenario_id", yaml_path.stem)
    odd_domain = meta.get("odd_cell", {}).get("domain", "open_sea")
    sim_settings = meta.get("simulation_settings", {})
    total_time = float(sim_settings.get("total_time", 600.0))

    os_init = data["ownShip"]["initial"]
    os_lat = os_init["position"]["latitude"]
    os_lon = os_init["position"]["longitude"]
    os_cog = float(os_init.get("cog", 0.0))
    os_sog = float(os_init.get("sog", 10.0))

    target_ships = data.get("targetShips", [])
    if not target_ships:
        return ScenarioBatchResult(sid, source, rule, odd_domain, False, 0.0, 0.0, 0.0,
                                   "no target ships")
    ts_init = target_ships[0]["initial"]
    ts_lat = ts_init["position"]["latitude"]
    ts_lon = ts_init["position"]["longitude"]
    ts_cog = float(ts_init.get("cog", 180.0))
    ts_sog = float(ts_init.get("sog", 10.0))

    t0 = time.perf_counter()
    min_cpa_nm, frames = _simulate_scenario_geometric(
        os_lat, os_lon, os_cog, os_sog,
        ts_lat, ts_lon, ts_cog, ts_sog,
        total_time_s=total_time,
    )
    wall_clock = time.perf_counter() - t0

    scorer = HagenScorer(cpa_target_nm=0.27)
    for f in frames:
        scorer.score_frame(
            own_lat=os_lat, own_lon=os_lon,
            own_heading=math.radians(os_cog), own_sog=os_sog,
            targets=[(ts_lat, ts_lon, ts_cog, ts_sog)],
            rule_states={rule: f["rule14"]},
            t_action_s=f["stamp"], t_target_action_s=300.0,
            rudder_deg=f["rudder_deg"], turning_rate_dps=2.0,
            behavior_phase=f["behavior_phase"],
            trajectory_curvature=0.0, trajectory_accel_ms2=0.0,
            applicable_rule=rule,
        )

    verdict = scorer.compute_verdict(min_cpa_nm, {rule: "full" if min_cpa_nm >= 0.27 else "violated"})
    _, quality = scorer.get_quality_score()

    return ScenarioBatchResult(
        scenario_id=sid, source=source, rule=rule,
        odd_domain=odd_domain, pass_fail=verdict,
        quality_score=quality, min_cpa_nm=min_cpa_nm,
        wall_clock_s=wall_clock,
    )

    return results
```

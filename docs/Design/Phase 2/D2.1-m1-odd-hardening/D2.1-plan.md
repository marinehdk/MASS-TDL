```cpp
EnvelopeState OddStateMachine::step(double score, double tdl_s, double tmr_s,
                                     EventFlags events, OddZoneHealthPair zone_health,
                                     std::chrono::steady_clock::time_point now) noexcept {
  // --- M7 Checker VETO (highest priority) ---
  if (events.m7_safety_critical || events.m7_safety_mrc_required) {
    current_ = EnvelopeState::MrCPrep;
    rationale_ = "M7 Safety Supervisor VETO — forced MrCPrep (Checker authority > Doer)";
    return current_;
  }

  // --- CRITICAL health → immediate MrCPrep ---
  if (zone_health.health == SystemHealth::Critical) {
    current_ = EnvelopeState::MrCPrep;
    rationale_ = "SystemHealth CRITICAL → immediate MrCPrep";
    return current_;
  }

  // --- DEGRADED health → apply stale factor ---
  double effective_score = score;
  if (zone_health.health == SystemHealth::Degraded) {
    effective_score *= params_.stale_degradation_factor;
  }

  // --- ODD_C zone → tightened threshold ---
  double edge_threshold = params_.in_to_edge;
  double out_threshold = params_.edge_to_out;
  if (zone_health.zone == OddZone::C) {
    out_threshold = 0.6;  // Harbor requires tighter margin

  return compute_next(effective_score, tdl_s, tmr_s, events, zone_health, now);
}
```

```cpp
ScoreTriple ConformanceScoreCalculator::compute_with_ema(
    const ScoringInputs& inputs,
    const ParameterSet& params,
    EmaState& ema_state,
    double dt_s) const noexcept {
  
  // Compute raw scores first
  ScoreTriple raw = compute(inputs, params);
  
  // NaN guard: if raw score is NaN, keep previous filtered value
  if (std::isnan(raw.conformance_score)) {
    return ScoreTriple{raw.e_score, raw.t_score, raw.h_score,
                       ema_state.filtered_value};
  }
  
  // EMA: filtered(t) = α × raw(t) + (1-α) × filtered(t-1)
  // α = dt / τ, clamped to [0, 1]
  double tau = params.ema_tau_s;
  if (tau <= 0.0) {
    // EMA disabled — return raw
    ema_state.filtered_value = raw.conformance_score;
    ema_state.tau_s = tau;
    return raw;
  }
  
  double alpha = dt_s / tau;
  if (alpha > 1.0) alpha = 1.0;
  if (alpha < 0.0) alpha = 0.0;
  
  ema_state.filtered_value = alpha * raw.conformance_score +
                              (1.0 - alpha) * ema_state.filtered_value;
  ema_state.tau_s = tau;
  
  // Return smoothed conformance score
  return ScoreTriple{raw.e_score, raw.t_score, raw.h_score,
                     ema_state.filtered_value};
}
```

```cpp
                                        const ParameterSet& params) const noexcept {
  auto idx = static_cast<int>(op_state);
  if (idx < 0 || idx >= 5) {
```

```cpp
/// ROT_max curve point (speed → max turn rate)
struct RotMaxCurvePoint {
  double speed_kn{0.0};
  double rot_max_deg_s{0.0};
};

/// Linear interpolation between ROT_max curve points
inline double interpolate_rot_max(double speed_kn,
                                   const std::vector<RotMaxCurvePoint>& curve) noexcept {
  if (curve.empty()) return 0.0;
  if (curve.size() == 1) return curve[0].rot_max_deg_s;
  
  // Below range
  if (speed_kn <= curve.front().speed_kn) return curve.front().rot_max_deg_s;
  // Above range
  if (speed_kn >= curve.back().speed_kn) return curve.back().rot_max_deg_s;
  
  // Linear interpolation between bracketing points
  for (size_t i = 0; i < curve.size() - 1; ++i) {
    if (speed_kn >= curve[i].speed_kn && speed_kn <= curve[i + 1].speed_kn) {
      double t = (speed_kn - curve[i].speed_kn) /
                 (curve[i + 1].speed_kn - curve[i].speed_kn);
      return curve[i].rot_max_deg_s +
             t * (curve[i + 1].rot_max_deg_s - curve[i].rot_max_deg_s);
    }
  }
  return curve.back().rot_max_deg_s;
}
```

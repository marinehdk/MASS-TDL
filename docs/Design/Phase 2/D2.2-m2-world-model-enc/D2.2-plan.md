```cpp
CpaUncertainty
CpaTcpaCalculator::propagate_ukf_(const Eigen::Vector2d& rel_pos,
                                   const Eigen::Vector2d& rel_vel,
                                   const Eigen::Matrix2d& sigma_rel_pos,
                                   const Eigen::Matrix2d& sigma_rel_vel) const {
  // ── State: x = [Δx, Δy, Δvx, Δvy]^T (4-dim) ──
  const int n = 4;  // state dimension
  Eigen::Vector4d x;
  x << rel_pos(0), rel_pos(1), rel_vel(0), rel_vel(1);

  // ── Covariance: block-diagonal of position and velocity ──
  Eigen::Matrix4d P = Eigen::Matrix4d::Zero();
  P.block<2, 2>(0, 0) = sigma_rel_pos;
  P.block<2, 2>(2, 2) = sigma_rel_vel;

  // ── Generate 2n+1 = 9 sigma points ──
  const double alpha = cfg_.ukf_alpha;
  const double beta = cfg_.ukf_beta;
  const double kappa = cfg_.ukf_kappa;
  const double lambda = alpha * alpha * (n + kappa) - n;
  const double w_m0 = lambda / (n + lambda);
  const double w_c0 = w_m0 + (1.0 - alpha * alpha + beta);

  Eigen::LLT<Eigen::Matrix4d> llt(P);
  if (llt.info() != Eigen::Success) {
    // Fallback: return linear propagation result
    return propagate_linear_(rel_pos, rel_vel, sigma_rel_pos, sigma_rel_vel);
  }
  Eigen::Matrix4d sqrt_P = llt.matrixL();
  double sqrt_nl = std::sqrt(n + lambda);

  // ── Nonlinear observation: h(x) = [CPA(x), TCPA(x)]^T ──
  auto h = [](const Eigen::Vector4d& xi) -> Eigen::Vector2d {
    double a = xi(2) * xi(2) + xi(3) * xi(3);  // Δvx² + Δvy²
    double b = 2.0 * (xi(0) * xi(2) + xi(1) * xi(3));
    double c = xi(0) * xi(0) + xi(1) * xi(1);

    Eigen::Vector2d y;
    if (a < 1e-12) {
      y(0) = std::sqrt(std::max(c, 0.0));  // CPA = current distance
      y(1) = std::numeric_limits<double>::infinity();
    } else {
      double t_cpa = -b / (2.0 * a);
      double cpa_sq = c - b * b / (4.0 * a);
      y(0) = std::sqrt(std::max(cpa_sq, 0.0));
      y(1) = std::max(t_cpa, 0.0);
    }
    return y;
  };

  // ── Propagate sigma points through h ──
  std::array<Eigen::Vector2d, 9> Y;
  Y[0] = h(x);  // center point

  for (int i = 0; i < n; ++i) {
    Eigen::Vector4d dx = sqrt_nl * sqrt_P.col(i);
    Y[2 * i + 1] = h(x + dx);
    Y[2 * i + 2] = h(x - dx);
  }

  // ── Recover mean and covariance of observation ──
  Eigen::Vector2d y_mean = w_m0 * Y[0];
  for (int i = 0; i < n; ++i) {
    double wi = 1.0 / (2.0 * (n + lambda));
    y_mean += wi * (Y[2 * i + 1] + Y[2 * i + 2]);
  }

  Eigen::Matrix2d Py = w_c0 * (Y[0] - y_mean) * (Y[0] - y_mean).transpose();
  for (int i = 0; i < n; ++i) {
    double wi = 1.0 / (2.0 * (n + lambda));
    Py += wi * (Y[2 * i + 1] - y_mean) * (Y[2 * i + 1] - y_mean).transpose();
    Py += wi * (Y[2 * i + 2] - y_mean) * (Y[2 * i + 2] - y_mean).transpose();
  }

  return CpaUncertainty{
      .cpa_sigma_m = std::sqrt(std::max(Py(0, 0), 0.0)),
      .tcpa_sigma_s = std::sqrt(std::max(Py(1, 1), 0.0))};
}
```

constexpr double kEarthRadiusM = 6371000.0;
constexpr double kDegToRad = M_PI / 180.0;

/// Simple point-in-bounding-box check.
bool in_bbox_(double lat, double lon,
              double lat_min, double lat_max,
              double lon_min, double lon_max) {
  return lat >= lat_min && lat <= lat_max &&
         lon >= lon_min && lon <= lon_max;
}

/// Haversine distance (m) between two WGS84 points.
double haversine_m_(double lat1, double lon1,
                    double lat2, double lon2) {
  double dlat = (lat2 - lat1) * kDegToRad;
  double dlon = (lon2 - lon1) * kDegToRad;
  double a = std::sin(dlat / 2.0) * std::sin(dlat / 2.0) +
             std::cos(lat1 * kDegToRad) * std::cos(lat2 * kDegToRad) *
                 std::sin(dlon / 2.0) * std::sin(dlon / 2.0);
  return 2.0 * kEarthRadiusM * std::atan2(std::sqrt(a), std::sqrt(1.0 - a));
}
}  // namespace

    if (!f.is_open()) return false;
    nlohmann::json j;
    f >> j;
    charts_.clear();

      EncChart c;
      c.name = chart.value("name", "unknown");
      c.lat_min = chart.value("lat_min", -90.0);
      c.lat_max = chart.value("lat_max", 90.0);
      c.lon_min = chart.value("lon_min", -180.0);
      c.lon_max = chart.value("lon_max", 180.0);
      c.has_tss = chart.value("has_tss", false);
      c.has_narrow_channel = chart.value("has_narrow_channel", false);

                                double horizon_radius_m) const {
  std::vector<std::string> zones;
  if (!loaded_) return zones;
  for (const auto& c : charts_) {
    if (!c.exclusion_zones_json.empty()) {
      // Simple bbox overlap check with virtual horizon circle

EnvSanityChecker::EnvSanityChecker(Config cfg) : cfg_(std::move(cfg)) {}

EnvSanityChecker::Result
EnvSanityChecker::validate(const ZoneSnapshot& snapshot,
                            std::chrono::steady_clock::time_point now) const {
  Result r;
  r.corrected_snapshot = snapshot;

  // 1. Visibility range check
  // (visibility stored indirectly — check via zone_type for v1.0)
  // Full implementation would read EnvironmentState.visibility

  // 2. Staleness check
  double age_s = std::chrono::duration<double>(now - snapshot.stamp).count();
  if (age_s > cfg_.staleness_max_s) {
    r.passed = false;
    r.failure_reason = "environment_state_stale_" + std::to_string(age_s) + "s";
    r.confidence_multiplier = 0.5;
    if (last_valid_snapshot_.has_value()) {
      r.corrected_snapshot = last_valid_snapshot_.value();
    }
  }

  // 3. Current speed range check
  if (snapshot.current_speed_kn > cfg_.current_speed_max_kn) {
    r.passed = false;
    r.failure_reason = "current_speed_out_of_range_" +
                       std::to_string(snapshot.current_speed_kn) + "kn";
    r.corrected_snapshot.current_speed_kn = 0.0;
    r.corrected_snapshot.current_direction_deg = 0.0;
    r.confidence_multiplier *= 0.8;
  }

  // 4. Zone transition check
  if (!last_zone_type_.empty()) {
    if (!check_zone_transition_(snapshot.zone_type, last_zone_type_)) {
      r.passed = false;
      r.failure_reason = "illegal_zone_transition_" + last_zone_type_ +
                         "_to_" + snapshot.zone_type;
      r.corrected_snapshot.zone_type = last_zone_type_;
      r.confidence_multiplier *= 0.7;
    }
  }

  // Cache valid snapshot
  if (r.passed) {
    last_valid_snapshot_ = snapshot;
  }
  last_zone_type_ = snapshot.zone_type;

  return r;
}

bool EnvSanityChecker::check_zone_transition_(
    const std::string& current, const std::string& previous) const {
  // OPEN → PORT without NARROW_CHANNEL is suspect
  if (previous == "open_water" && current == "port") {
    return false;
  }
  // Allow all other transitions for v1.0
  return true;
}

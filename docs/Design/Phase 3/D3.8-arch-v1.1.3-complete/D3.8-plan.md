```bash
#!/usr/bin/env bash
# scripts/gen-pdf.sh — 架构报告 v1.1.3 PDF 生成
# 依赖：pandoc ≥ 3.0 + texlive-xetex + Noto CJK 字体
# macOS 安装：brew install --cask font-noto-serif-cjk && brew install basictex
#              sudo tlmgr install xecjk ctex
# 用法：bash scripts/gen-pdf.sh [output-dir]

set -euo pipefail

SRC="docs/Design/Architecture Design/MASS_ADAS_L3_TDL_架构设计报告.md"
OUT="${1:-dist}/MASS_ADAS_L3_TDL_架构设计报告_v1.1.3.pdf"
mkdir -p "$(dirname "$OUT")"

pandoc "$SRC" \
  -o "$OUT" \
  --pdf-engine=xelatex \
  --toc \
  --toc-depth=3 \
  -V documentclass=article \
  -V geometry:margin=2.5cm \
  -V mainfont="Noto Serif CJK SC" \
  -V CJKmainfont="Noto Serif CJK SC" \
  -V fontsize=11pt \
  -V colorlinks=true \
  -V linkcolor=blue \
  --highlight-style=tango

echo "PDF 生成完成：$OUT"
```
